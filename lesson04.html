NEIS_URL = "https://open.neis.go.kr/hub"
TEACHER_CSV = (

    "main/data/danggok_lunch_202610.csv"

MENU_BREAK = re.compile(r"<br\s*/?\s*>", re.IGNORECASE)
# 메뉴 끝의 (5.6.13.), (1, 2) 같은 번호 묶음만 제거한다.
ALLERGY_SUFFIX = re.compile(r"\s*\(\s*\d+(?:[\s.,·]+\d+)*[\s.,·]*\)\s*$")


class LookupErrorNEIS(Exception):
    pass


def read_api_key():
    try:
        return st.secrets.get("NEIS_API_KEY", "").strip()
    except StreamlitSecretNotFoundError:
        return ""


def last_month():
    first_of_this_month = datetime.now(ZoneInfo("Asia/Seoul")).date().replace(day=1)
    end = first_of_this_month - timedelta(days=1)
    return end.replace(day=1), end


def check_result(result):
    code = result.get("CODE", "")
    if code == "INFO-200":
        return False
    if code != "INFO-000":
        raise LookupErrorNEIS(f"나이스 응답 코드를 확인해 주세요 ({code or '코드 없음'}).")
    return True


@st.cache_data(ttl=600, show_spinner=False)
def fetch_neis(service, filters, api_key):
    page_size = 100 if api_key else 5
    params = {"Type": "json", "pSize": page_size, "pIndex": 1, **filters}
    if api_key:
        params["KEY"] = api_key

    rows = []
    while True:
        try:
            response = requests.get(f"{NEIS_URL}/{service}", params=params, timeout=12)
            response.raise_for_status()
            data = response.json()
        except (requests.RequestException, ValueError):
            # 요청 주소에 KEY가 포함될 수 있어 예외 원문은 화면에 표시하지 않는다.
            raise LookupErrorNEIS("나이스 서버에 연결하거나 응답을 읽는 데 실패했어요.") from None

        try:
            if "RESULT" in data:
                if not check_result(data["RESULT"]) and not rows:
                    return [], 0
                raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            head = data[service][0]["head"]
            for item in head:
                if "RESULT" in item and not check_result(item["RESULT"]):
                    if not rows:
                        return [], 0
                    raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            total = int(
                next(item["list_total_count"] for item in head if "list_total_count" in item)
            )
            page = data[service][1]["row"]
            if not isinstance(page, list) or (not page and len(rows) < total):
                raise LookupErrorNEIS("나이스가 보낸 목록이 불완전해요.")
        except (KeyError, IndexError, TypeError, ValueError, StopIteration):
            raise LookupErrorNEIS("나이스 응답에서 목록이나 전체 건수를 읽을 수 없어요.") from None

        rows.extend(page)
        if not api_key or len(rows) >= total:
            return rows, total
        params["pIndex"] += 1


@st.cache_data(ttl=600, show_spinner=False)
def load_teacher_lunch():
    response = requests.get(TEACHER_CSV, timeout=12)
    response.raise_for_status()
    reader = csv.DictReader(StringIO(response.content.decode("utf-8-sig")))
    required = {"MLSV_YMD", "MMEAL_SC_NM", "DDISH_NM"}
    if not required.issubset(reader.fieldnames or []):
        raise ValueError("필수 열 누락")
    return [row for row in reader if row["MMEAL_SC_NM"].strip() == "중식"]


def show_ranking(by_date, total, year, month):
    dates_by_dish = defaultdict(set)
    for day, dishes in by_date.items():
        for dish in dishes:
            name = ALLERGY_SUFFIX.sub("", dish).strip()
            if name:
                dates_by_dish[name].add(day)

    if not dates_by_dish:
        st.info("순위를 집계할 메뉴 이름이 없어요.")
        return

    excluded = st.multiselect(
        "반찬 순위에서 뺄 메뉴",
        options=sorted(dates_by_dish),
        help="우리 반에서 빼고 싶은 밥, 국 등의 메뉴를 골라 주세요.",
    )
    included = dates_by_dish.keys() - set(excluded)
    if not included:
        st.info("모든 메뉴를 제외했어요. 제외 목록에서 메뉴를 하나 빼면 순위가 보여요.")
        return

    ranked = sorted(included, key=lambda name: (-len(dates_by_dish[name]), name))
    max_days = len(dates_by_dish[ranked[0]])
    winners = [name for name in ranked if len(dates_by_dish[name]) == max_days]
    first_place = "공동 1위" if len(winners) > 1 else "1위"
    st.write(f"{first_place} · {max_days}일 · {', '.join(winners)}")

    title = (
        "조회된 급식일의 반찬 순위"
        if len(by_date) < total
        else f"{year}년 {month}월 반찬 순위"
    )
    top_ten = ranked[:10]
    chart_rows = [
        {"메뉴": name, "일수": len(dates_by_dish[name]), "표시": f"{len(dates_by_dish[name])}일"}
        for name in top_ten
    ]
    base = alt.Chart(alt.Data(values=chart_rows)).encode(
        y=alt.Y("메뉴:N", sort=top_ten, title=None),
        x=alt.X(
            "일수:Q",
            title="나온 급식일 수",
            axis=alt.Axis(tickMinStep=1, format="d"),
            scale=alt.Scale(domain=[0, max_days + max(1, max_days * 0.2)]),
        ),
        tooltip=["메뉴:N", "일수:Q"],
    )
    bars = base.mark_bar()
    labels = base.mark_text(align="left", baseline="middle", dx=5).encode(text="표시:N")
    chart = (bars + labels).properties(title=title, height=32 * len(top_ten))
    st.altair_chart(chart, use_container_width=True)

    selected = st.selectbox("나온 날짜를 확인할 반찬", options=ranked)
    days = sorted(dates_by_dish[selected])
    st.write(f"{selected} · 총 {len(days)}일")
    st.write(", ".join(day.isoformat() for day in days))


def show_menu(rows, count_label, total, year, month):
    by_date = defaultdict(list)
    for row in rows:
        day = datetime.strptime(row["MLSV_YMD"].strip(), "%Y%m%d").date()
        dishes = MENU_BREAK.split(row["DDISH_NM"])
        by_date[day].extend(dish.strip() for dish in dishes if dish.strip())

    st.write(f"조회된 날짜 수 {len(by_date)}일 · {count_label} {total}건")
    if not by_date:
        st.info("이 기간에 등록된 중식이 없어요.")
        return

    table = [
        {"날짜": day.isoformat(), "메뉴": " · ".join(by_date[day])}
        for day in sorted(by_date)
    ]
    st.dataframe(table, hide_index=True, use_container_width=True)
    show_ranking(by_date, total, year, month)

친구가 앱에서 무엇을 하는지 직접 보고 기록하는 일을 디자인 씽킹에서는 사용자 관찰이라고 합니다. 예를 들어 월을 고르는 칸이 있는 앱에서 “월 선택을 두 번 지나쳤다”는 눈으로 확인한 행동입니다. “복잡해 보였다”는 그 행동에 대한 내 해석입니다. 우리 앱에서는 친구가 실제로 지나친 항목이나 다시 읽은 문구를 적으세요.

친구가 “어디에 며칠 나왔는지 쓰여 있어?”라고 물으면 그 말을 그대로 적으세요. 물어본 뒤 혼자 찾을 수도 있으므로 질문과 도움 여부는 따로 기록하세요. 처음 본 곳이 분명하지 않으면 과제가 끝난 뒤 “어디를 먼저 봤어요?”라고 묻고, 친구의 대답임을 표시하세요.

    • 수정 전에 적은 1위 반찬 이름과 제공 일수가 일치합니까? 공동 1위가 있었다면 모든 이름이 다시 나옵니까?
    • 1위 반찬을 선택해 나온 날짜를 세어 보세요. 같은 날의 같은 반찬은 한 번만 세었습니까? 서로 다른 날짜의 수가 막대 끝의 일수와 일치합니까?

    선생님 준비 자료로 확인할 때는 제외 메뉴를 비운 상태에서 당곡고등학교 · 2026년 10월, 조회된 날짜 수 20일, 1위 배추김치 9일을 확인할 수 있습니다. 나이스로 다른 달을 조회했다면 그달의 수정 전 기록과 비교하세요.

NEIS_URL = "https://open.neis.go.kr/hub"
TEACHER_CSV = (

    "main/data/danggok_lunch_202610.csv"

MENU_BREAK = re.compile(r"<br\s*/?\s*>", re.IGNORECASE)
# 메뉴 끝의 (5.6.13.), (1, 2) 같은 번호 묶음만 제거한다.
ALLERGY_SUFFIX = re.compile(r"\s*\(\s*\d+(?:[\s.,·]+\d+)*[\s.,·]*\)\s*$")


class LookupErrorNEIS(Exception):
    pass


def read_api_key():
    try:
        return st.secrets.get("NEIS_API_KEY", "").strip()
    except StreamlitSecretNotFoundError:
        return ""


def last_month():
    first_of_this_month = datetime.now(ZoneInfo("Asia/Seoul")).date().replace(day=1)
    end = first_of_this_month - timedelta(days=1)
    return end.replace(day=1), end


def check_result(result):
    code = result.get("CODE", "")
    if code == "INFO-200":
        return False
    if code != "INFO-000":
        raise LookupErrorNEIS(f"나이스 응답 코드를 확인해 주세요 ({code or '코드 없음'}).")
    return True


@st.cache_data(ttl=600, show_spinner=False)
def fetch_neis(service, filters, api_key):
    page_size = 100 if api_key else 5
    params = {"Type": "json", "pSize": page_size, "pIndex": 1, **filters}
    if api_key:
        params["KEY"] = api_key

    rows = []
    while True:
        try:
            response = requests.get(f"{NEIS_URL}/{service}", params=params, timeout=12)
            response.raise_for_status()
            data = response.json()
        except (requests.RequestException, ValueError):
            # 요청 주소에 KEY가 포함될 수 있어 예외 원문은 화면에 표시하지 않는다.
            raise LookupErrorNEIS("나이스 서버에 연결하거나 응답을 읽는 데 실패했어요.") from None

        try:
            if "RESULT" in data:
                if not check_result(data["RESULT"]) and not rows:
                    return [], 0
                raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            head = data[service][0]["head"]
            for item in head:
                if "RESULT" in item and not check_result(item["RESULT"]):
                    if not rows:
                        return [], 0
                    raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            total = int(
                next(item["list_total_count"] for item in head if "list_total_count" in item)
            )
            page = data[service][1]["row"]
            if not isinstance(page, list) or (not page and len(rows) < total):
                raise LookupErrorNEIS("나이스가 보낸 목록이 불완전해요.")
        except (KeyError, IndexError, TypeError, ValueError, StopIteration):
            raise LookupErrorNEIS("나이스 응답에서 목록이나 전체 건수를 읽을 수 없어요.") from None

        rows.extend(page)
        if not api_key or len(rows) >= total:
            return rows, total
        params["pIndex"] += 1


@st.cache_data(ttl=600, show_spinner=False)
def load_teacher_lunch():
    response = requests.get(TEACHER_CSV, timeout=12)
    response.raise_for_status()
    reader = csv.DictReader(StringIO(response.content.decode("utf-8-sig")))
    required = {"MLSV_YMD", "MMEAL_SC_NM", "DDISH_NM"}
    if not required.issubset(reader.fieldnames or []):
        raise ValueError("필수 열 누락")
    return [row for row in reader if row["MMEAL_SC_NM"].strip() == "중식"]


def show_ranking(by_date, total, year, month):
    dates_by_dish = defaultdict(set)
    for day, dishes in by_date.items():
        for dish in dishes:
            name = ALLERGY_SUFFIX.sub("", dish).strip()
            if name:
                dates_by_dish[name].add(day)

    if not dates_by_dish:
        st.info("순위를 집계할 메뉴 이름이 없어요.")
        return

    excluded = st.multiselect(
        "반찬 순위에서 뺄 메뉴",
        options=sorted(dates_by_dish),
        help="우리 반에서 빼고 싶은 밥, 국 등의 메뉴를 골라 주세요.",
    )
    included = dates_by_dish.keys() - set(excluded)
    if not included:
        st.info("모든 메뉴를 제외했어요. 제외 목록에서 메뉴를 하나 빼면 순위가 보여요.")
        return

    ranked = sorted(included, key=lambda name: (-len(dates_by_dish[name]), name))
    max_days = len(dates_by_dish[ranked[0]])
    winners = [name for name in ranked if len(dates_by_dish[name]) == max_days]
    first_place = "공동 1위" if len(winners) > 1 else "1위"
    st.write(f"{first_place} · {max_days}일 · {', '.join(winners)}")

    title = (
        "조회된 급식일의 반찬 순위"
        if len(by_date) < total
        else f"{year}년 {month}월 반찬 순위"
    )
    top_ten = ranked[:10]
    chart_rows = [
        {"메뉴": name, "일수": len(dates_by_dish[name]), "표시": f"{len(dates_by_dish[name])}일"}
        for name in top_ten
    ]
    base = alt.Chart(alt.Data(values=chart_rows)).encode(
        y=alt.Y("메뉴:N", sort=top_ten, title=None),
        x=alt.X(
            "일수:Q",
            title="나온 급식일 수",
            axis=alt.Axis(tickMinStep=1, format="d"),
            scale=alt.Scale(domain=[0, max_days + max(1, max_days * 0.2)]),
        ),
        tooltip=["메뉴:N", "일수:Q"],
    )
    bars = base.mark_bar()
    labels = base.mark_text(align="left", baseline="middle", dx=5).encode(text="표시:N")
    chart = (bars + labels).properties(title=title, height=32 * len(top_ten))
    st.altair_chart(chart, use_container_width=True)

    selected = st.selectbox("나온 날짜를 확인할 반찬", options=ranked)
    days = sorted(dates_by_dish[selected])
    st.write(f"{selected} · 총 {len(days)}일")
    st.write(", ".join(day.isoformat() for day in days))


def show_menu(rows, count_label, total, year, month):
    by_date = defaultdict(list)
    for row in rows:
        day = datetime.strptime(row["MLSV_YMD"].strip(), "%Y%m%d").date()
        dishes = MENU_BREAK.split(row["DDISH_NM"])
        by_date[day].extend(dish.strip() for dish in dishes if dish.strip())

    st.write(f"조회된 날짜 수 {len(by_date)}일 · {count_label} {total}건")
    if not by_date:
        st.info("이 기간에 등록된 중식이 없어요.")
        return

    st.subheader("가장 자주 나온 반찬")
    show_ranking(by_date, total, year, month)

    table = [
        {"날짜": day.isoformat(), "메뉴": " · ".join(by_date[day])}
        for day in sorted(by_date)
    ]
    st.dataframe(table, hide_index=True, use_container_width=True)

4. 새 친구에게 같은 과제로 다시 부탁하기

첫 번째 친구가 찾는 모습을 보지 않은 친구에게 부탁하세요. 처음과 같은 기기에서 같은 학교, 기간, 제외 메뉴를 선택하고 조회 결과의 맨 위를 보여 주세요. 1절에서 읽은 과제 문장을 그대로 읽어 주세요. 어느 곳을 고쳤는지는 과제가 끝난 뒤에 알려 주세요.

수정 전 칸에는 첫 번째 친구의 기록을 옮기고, 수정 후 칸에는 새 친구의 행동을 적으세요. 답을 찾지 못한 경우에도 마지막으로 본 화면을 적으면 두 친구가 어디까지 확인했는지 비교할 수 있습니다.

두 기록을 읽고 “첫 번째 친구는 어디에서 멈췄습니까? 새 친구는 어디에서 답을 확인했습니까?”에 답하세요. 새 친구도 같은 곳에서 멈췄다면 그 행동을 기록하고, 다음에 확인할 문구나 위치를 하나 적으세요.

기준 코드로 만든 앱은 급식실 열기를 눌러야 자료 선택과 학교 검색이 표시됩니다. 관찰을 준비하는 사람이 먼저 버튼을 누르세요. ‘키 없이 최대 5건만 보입니다’라는 안내가 나오거나 나이스 조회가 막히면 선생님 준비 자료 불러오기를 선택하고, 화면에 ‘당곡고등학교 · 2026년 10월’과 ‘15일’이 나오는지 확인하세요.

AI가 만든 그래프 모양이 기준 코드와 다릅니다

AI가 고른 그래프 도구에 따라 글꼴이나 막대 모양은 달라질 수 있습니다. 채운 예를 사용했다면 소제목을 먼저 읽고 아래로 내려 보세요. 그래프 다음에 날짜별 중식 표가 나오는지, 막대에 적힌 일수가 수정 전 값과 일치하는지 확인하세요.

화면을 고친 뒤 1위 반찬이나 일수가 달라졌습니다

수정 전 기록에서 자료, 학교, 연월, 제외 메뉴를 하나씩 확인하세요. 특히 선생님 준비 자료와 나이스 조회 자료는 기간이 다를 수 있습니다. 조건을 맞춘 뒤에도 값이 다르면 1위 반찬을 선택하고 나온 날짜 목록을 식단 표와 대조하세요.

AI 딴소리 멈추기 한 줄

그래프 위치를 옮겨 달라고 했는데 반찬을 세는 방법까지 달라졌다면, 수정 전 코드를 붙이고 다음 문장을 보내세요.

수정 범위 다시 알려 주기
아래 main.py에서 반찬을 세는 기준과 제외 메뉴 선택은 그대로 두고, 내가 요청한 화면 한 곳만 고쳐서 main.py 전체를 줘.

오늘 정리

1차시에 배포한 첫 화면에 학교 검색과 중식 조회를 더하고, 날짜별 식단으로 반찬 순위를 구했습니다. 오늘은 처음 보는 친구가 그 순위를 어떻게 찾는지 확인해 1~4차시 공통 대시보드를 완성했습니다.

비교표를 보며 친구가 한 행동, 그 행동을 보고 고친 곳, 새 친구가 도움을 받았는지를 차례로 말하세요. 마지막에 같은 조건에서 확인한 1위와 제공 일수를 덧붙이세요.

확인 문제

문제 1 친구가 반찬 1위를 찾는 모습을 관찰했습니다. 기록표의 ‘멈춘 곳’과 ‘물어본 말’에 옮길 수 있는 문장을 고르세요.
손을 멈춘 곳은 볼 수 있고, 친구의 질문은 들은 대로 적을 수 있습니다. 왜 멈췄는지 알고 싶으면 과제가 끝난 뒤 친구에게 물어보세요.
문제 2 순위 그래프를 표 위로 옮겼습니다. 새 친구가 1위를 혼자 찾을 수 있는지 어떻게 확인하면 좋겠습니까?
두 친구에게 같은 과제를 주면 어디를 먼저 보았고 어디에서 도움을 받았는지 비교할 수 있습니다. 위치나 반찬 이름을 미리 알려 주면 새 친구는 그 단서를 사용하게 됩니다.
문제 3 같은 준비 자료와 제외 메뉴로 확인했더니 배추김치가 수정 전에는 9일, 수정 후에는 10일로 표시됩니다. 먼저 무엇을 확인해야 합니까?
제공 일수는 그 반찬이 나온 서로 다른 날짜의 수입니다. 순위가 같아도 일수가 달라졌다면 어떤 날짜를 세었는지 확인해야 합니다.
