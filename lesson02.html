1. API로 급식을 어떻게 조회합니까?

나이스 홈페이지에서는 사람이 학교를 고르고 식단을 읽습니다. 우리가 만든 앱도 나이스에서 식단을 받을 수 있습니다. 이때 사용하는 API는 프로그램끼리 자료를 주고받는 방법을 정한 약속입니다. 앱이 정해진 주소로 학교와 기간 같은 조건을 보내면 나이스는 약속된 형식으로 결과를 돌려줍니다. 오늘 앱은 두 번에 걸쳐 필요한 자료를 찾습니다.

조회 순서앱이 보내는 조건응답에서 읽을 내용
먼저 학교를 찾습니다.학교 이름을 보냅니다. 이름의 일부로도 검색할 수 있습니다.학교명과 지역을 보고 학교를 고릅니다. 앱은 선택한 학교의 교육청 코드와 학교 코드를 읽습니다.
그 학교의 점심을 찾습니다.두 코드와 식사 종류인 중식, 한국 시간 기준 지난달 1일부터 말일까지의 날짜를 보냅니다.조건에 맞는 전체 건수와 이번에 받은 식단 목록을 읽습니다. 식단에는 급식일과 그날의 메뉴가 들어 있습니다.

    import csv
import re
from collections import defaultdict
from datetime import datetime, timedelta
from io import StringIO
from zoneinfo import ZoneInfo

import requests
import streamlit as st
from streamlit.errors import StreamlitSecretNotFoundError


NEIS_URL = "https://open.neis.go.kr/hub"
TEACHER_CSV = (
    "https://raw.githubusercontent.com/greatsong/dugeun-lunch-claude/"
    "main/data/danggok_lunch_202609.csv"
)


class LookupErrorNEIS(Exception):
    pass


def read_api_key():
    try:
        return st.secrets.get("NEIS_API_KEY", "").strip()
    except StreamlitSecretNotFoundError:
        return ""


def last_month():
    first_of_this_month = datetime.now(ZoneInfo("Asia/Seoul")).date().replace(day=1)
    end = first_of_this_month - timedelta(days=1)
    return end.replace(day=1), end


def check_result(result):
    code = result.get("CODE", "")
    if code == "INFO-200":
        return False
    if code != "INFO-000":
        raise LookupErrorNEIS(f"나이스 응답 코드를 확인해 주세요 ({code or '코드 없음'}).")
    return True


@st.cache_data(ttl=600, show_spinner=False)
def fetch_neis(service, filters, api_key):
    page_size = 100 if api_key else 5
    params = {"Type": "json", "pSize": page_size, "pIndex": 1, **filters}
    if api_key:
        params["KEY"] = api_key

    rows = []
    while True:
        try:
            response = requests.get(f"{NEIS_URL}/{service}", params=params, timeout=12)
            response.raise_for_status()
            data = response.json()
        except (requests.RequestException, ValueError):
            # 요청 주소에 KEY가 포함될 수 있어 예외 원문은 화면에 표시하지 않는다.
            raise LookupErrorNEIS("나이스 서버에 연결하거나 응답을 읽는 데 실패했어요.") from None

        try:
            if "RESULT" in data:
                if not check_result(data["RESULT"]) and not rows:
                    return [], 0
                raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            head = data[service][0]["head"]
            for item in head:
                if "RESULT" in item and not check_result(item["RESULT"]):
                    if not rows:
                        return [], 0
                    raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            total = int(next(item["list_total_count"] for item in head if "list_total_count" in item))
            page = data[service][1]["row"]
            if not isinstance(page, list) or (not page and len(rows) < total):
                raise LookupErrorNEIS("나이스가 보낸 목록이 불완전해요.")
        except (KeyError, IndexError, TypeError, ValueError, StopIteration):
            raise LookupErrorNEIS("나이스 응답에서 목록이나 전체 건수를 읽을 수 없어요.") from None

        rows.extend(page)
        if not api_key or len(rows) >= total:
            return rows, total
        params["pIndex"] += 1


@st.cache_data(ttl=600, show_spinner=False)
def load_teacher_lunch():
    response = requests.get(TEACHER_CSV, timeout=12)
    response.raise_for_status()
    reader = csv.DictReader(StringIO(response.content.decode("utf-8-sig")))
    required = {"MLSV_YMD", "MMEAL_SC_NM", "DDISH_NM"}
    if not required.issubset(reader.fieldnames or []):
        raise ValueError("필수 열 누락")
    return [row for row in reader if row["MMEAL_SC_NM"].strip() == "중식"]


def show_menu(rows, count_label, total):
    by_date = defaultdict(list)
    for row in rows:
        day = datetime.strptime(row["MLSV_YMD"].strip(), "%Y%m%d").date()
        dishes = re.split(r"<br\s*/?\s*>", row["DDISH_NM"], flags=re.IGNORECASE)
        by_date[day].extend(dish.strip() for dish in dishes if dish.strip())

    st.write(f"조회된 날짜 수 {len(by_date)}일 · {count_label} {total}건")
    if not by_date:
        st.info("이 기간에 등록된 중식이 없어요.")
        return

    table = [
        {"날짜": day.isoformat(), "메뉴": " · ".join(by_date[day])}
        for day in sorted(by_date)
    ]
    st.dataframe(table, hide_index=True, use_container_width=True)


st.title("두근두근 급식실")
if st.button("급식실 열기"):
    st.session_state["lunch_open"] = True

if not st.session_state.get("lunch_open"):
    st.stop()

st.write("우리 학교 급식실에 오신 걸 환영합니다!")
api_key = read_api_key()
if not api_key:
    st.info("키 없이 최대 5건만 보입니다")

source = st.radio("자료 선택", ["나이스에서 조회", "선생님 준비 자료 불러오기"])

if source == "선생님 준비 자료 불러오기":
    st.subheader("당곡고등학교 · 2026년 9월 · 선생님 준비 자료")
    try:
        lunches = load_teacher_lunch()
        show_menu(lunches, "준비 자료 중식 행 수", len(lunches))
    except (requests.RequestException, csv.Error, ValueError, KeyError, TypeError, AttributeError):
        st.error("선생님 준비 자료를 불러오거나 읽을 수 없어요.")
    st.stop()

query = st.text_input("학교 이름", value="당곡고등학교").strip()
if not query:
    st.info("검색할 학교 이름을 입력해 주세요.")
    st.stop()

try:
    schools, school_total = fetch_neis("schoolInfo", {"SCHUL_NM": query}, api_key)
    if not schools:
        st.info("이 이름으로 찾은 학교가 없어요. 학교 이름의 일부만 입력해 보세요.")
        st.stop()
    if len(schools) < school_total:

3. 인증키를 Secrets에 넣으세요

우리 학교의 지난달 중식 전체를 조회할 때는 인증키를 사용합니다. 키는 선생님이 안내한 방식으로 받으세요. 실제 키를 입력할 곳은 배포한 앱의 Secrets 설정입니다. 키를 코드, 공개 앱 화면, AI 채팅에 붙이지 마세요.

  1. 배포한 앱의 설정을 여세요

    Streamlit Community Cloud에 로그인하고 내 앱 목록에서 지난 시간에 배포한 앱을 찾으세요. 해당 앱의 메뉴에서 Settings를 열고 Secrets를 선택하세요.

  2. Secrets 칸에 한 줄을 추가하세요

    아래 한 줄을 Secrets 칸에 복사하세요. 곧은 큰따옴표 두 개는 남기고 그 사이의 …를 받은 키 전체로 바꾸세요. Save changes 또는 저장 버튼을 누르세요.

    앱 설정의 Secrets 칸
    NEIS_API_KEY = "…"
  3. 같은 학교, 같은 달로 다시 조회하세요

    앱을 다시 열고 급식실 열기를 누르세요. 나이스에서 조회를 고르고 아까와 같은 학교와 지역을 선택하세요. ‘키 없이 최대 5건만 보입니다’ 안내가 사라졌는지, 식단 표가 나오는지 확인하세요. 조회 전후의 날짜 수와 전체 건수를 각각 비교하세요.

키를 받지 못했거나 조회 오류가 계속되면 선생님 준비 자료 불러오기를 선택하세요. 이때도 표를 읽고 다음 실습을 할 수 있습니다. 기록에는 화면에 표시된 학교, 월, 자료 출처를 그대로 쓰세요.

4. 몇 건을 받았는지 판단하세요

“표가 다섯 줄이면 지난달에 점심을 제공한 날도 5일입니까?”
두근의 질문 다섯 줄은 화면에서 셀 수 있습니다. 지난달 중식을 모두 받았다고 말하려면 어느 수를 더 확인해야 합니까?

조회된 날짜 수는 받은 목록에 서로 다른 급식일이 며칠 있는지 나타냅니다. 받은 행 수는 날짜별로 묶기 전 식단 목록의 길이입니다. 응답의 전체 건수는 선택한 학교와 조회 기간, 중식이라는 조건에 맞는 행이 나이스에 모두 몇 건 있는지 나타냅니다. 한 달치를 모두 받았는지 판단할 때는 받은 행 수와 전체 건수를 비교하세요.

화면에서 확인한 내용기록할 판단
나이스에서 키 없이 조회했습니다. 예를 들어 서로 다른 날짜가 5일이고 응답의 전체 건수가 18건입니다.키 없이 최대 5행을 받을 수 있고 날짜가 5개이므로 받은 행은 5건입니다. ‘전체 18건 중 5건을 받았습니다’라고 기록합니다.
나이스에서 키로 조회했습니다. 예를 들어 서로 다른 날짜가 18일이고 응답의 전체 건수도 18건입니다.서로 다른 18일이 모두 표에 있으므로 전체 18건을 확인했습니다. ‘전체 18건 중 18건을 받았습니다’라고 기록합니다.
준비 자료를 골랐습니다. 조회된 날짜 수는 20일이고 준비 자료 중식 행 수는 20건입니다.‘당곡고등학교 · 2026년 9월 · 선생님 준비 자료, 파일의 전체 20건 중 20건을 받았습니다’라고 기록합니다.

기준 코드는 같은 날짜의 식단을 한 행에 모아서 보여 줍니다. 같은 날짜에 원래 자료가 두 행 있었다면 표에는 한 줄로 보입니다. 나이스 조회에서 날짜 수와 전체 건수가 다르고 받은 행 수를 판단하기 어렵다면, 같은 AI 대화에 아래 요청을 보내세요. 수정한 앱을 열어 같은 조건으로 조회한 뒤 두 건수를 읽으세요.

받은 행 수를 따로 확인할 때
나이스에서 실제로 받은 식단이 날짜별로 묶기 전에 몇 행인지 세어서 ‘받은 행 수’로 보여 줘. 조회된 날짜 수, 받은 행 수, 응답의 전체 건수를 표 위에서 함께 읽을 수 있게 해 줘.

내가 조회한 결과를 기록하세요

표 위의 제목을 먼저 읽고 아래 빈칸을 수업 기록에 채우세요. 위 표의 18건은 설명을 위한 예입니다. 내 기록에는 실제로 확인한 수를 쓰세요. 인증이나 통신 오류로 표를 받지 못했다면 건수는 비워 두고 오류 문구를 적으세요.

조회 기록
학교와 지역 □ · 조회한 연도와 월 □ · 자료 출처 □
조회된 급식일은 □일입니다. 전체 □건 중 □건을 받았습니다.
이 자료로 확인할 수 있는 기간은 □입니다.

막힐 때 펼쳐 보세요

‘이 기간에 등록된 중식이 없어요’ 또는 INFO-200이 나옵니다

급식 조회에서 INFO-200은 보낸 조건에 맞는 식단이 없다는 뜻입니다. 선택한 학교의 이름과 지역, 표 위의 연도와 월이 맞는지 확인하세요. 조건이 맞으면 ‘해당 기간에 등록된 중식 없음’으로 기록하세요. 인증 오류나 서버에 연결할 수 없다는 안내가 나온 경우에는 급식 건수를 알 수 없습니다. 선생님 준비 자료를 선택해 실습을 이어 가세요.

지난달을 조회했는데 5건만 보입니다

표 위에서 키 없이 조회했다는 안내와 응답의 전체 건수를 찾으세요. 키 없이 받을 수 있는 자료는 처음 5건까지입니다. 전체가 5건보다 많아도 다음 쪽을 요청하면 같은 자료를 받습니다. 3절의 Secrets 설정을 마치고 다시 조회하거나 선생님 준비 자료를 고르세요.

목록에서 우리 학교를 찾을 수 없습니다

‘이 이름으로 찾은 학교가 없어요’가 나오면 띄어쓰기와 학교 이름을 확인한 뒤, 학교 이름의 일부로 다시 검색하세요. 반대로 여러 학교가 검색되고 일부만 표시된다는 안내가 나오면 ‘당곡’ 대신 ‘당곡고등학교’처럼 더 자세히 입력하세요. 목록에서 학교명과 지역을 함께 확인하세요.

Secrets에 저장한 뒤에도 키가 없다는 안내가 나옵니다

지금 열어 둔 앱과 Secrets를 저장한 앱이 같은지 확인하세요. 설정에서 NEIS_API_KEY의 대문자와 밑줄, 큰따옴표 안에 입력한 값을 직접 확인하고 다시 저장하세요. 변경이 반영된 뒤 앱을 다시 여세요. 계속 이전 화면이 보이면 Community Cloud의 해당 앱 메뉴에서 Reboot app을 선택한 뒤 다시 조회하세요.

키가 없다는 안내는 사라지고 인증 오류가 나오면 앱이 키를 읽었더라도 나이스가 그 키를 받아들이지 않은 경우입니다. 선생님께는 오류 코드만 알려 주세요. 키를 확인하는 동안에는 선생님 준비 자료를 사용하세요.

AI 딴소리 멈추기 한 줄

AI가 반찬 순위나 평가 버튼까지 추가했다면 같은 대화에 다음 한 줄을 보내세요.

이번 시간에 확인할 기능
첫 화면을 유지하고 학교 검색, 지난달 중식 조회, 선생님 준비 자료 선택, 날짜별 표와 건수 확인까지만 할 수 있게 main.py와 requirements.txt를 다시 보여 줘.

오늘 정리

나이스에서 학교를 찾을 때에는 학교명과 지역을 확인했습니다. 그 학교의 중식을 조회할 때에는 교육청 코드, 학교 코드, 식사 종류, 기간을 사용했습니다. 인증키는 배포한 앱의 Secrets에 저장하고, 조회 결과는 공개 화면에서 확인했습니다.

짝에게 어느 학교의 어느 달 자료를 몇 건 받았는지 말해 보세요. 자료 일부를 받았다면 확인한 날짜가 며칠인지 덧붙이세요. 준비 자료를 사용했다면 ‘당곡고등학교 2026년 9월 중식 20일’이라고 범위를 밝혀 주세요.

키 없이 받을 수 있는 식단은 최대 5건입니다. 표에 서로 다른 5일이 있으므로 5건을 받았습니다. 나이스가 알려 준 전체 19건까지 확인하려면 인증키로 다시 조회해야 합니다.
문제 2 선생님이 안내한 방식으로 실제 인증키를 받았습니다. 어디에 입력해야 합니까?

준비 자료를 고르면 앱이 정해진 파일을 읽습니다. 표 위의 ‘당곡고등학교 · 2026년 9월 · 선생님 준비 자료’에서 학교와 기간, 출처를 확인할 수 있습니다.
