from collections import defaultdict
from datetime import datetime, timedelta
from io import StringIO
from zoneinfo import ZoneInfo

import requests
import streamlit as st
from streamlit.errors import StreamlitSecretNotFoundError


NEIS_URL = "https://open.neis.go.kr/hub"
TEACHER_CSV = (
    "https://raw.githubusercontent.com/greatsong/dugeun-lunch-claude/"
    "main/data/danggok_lunch_202610.csv"

class LookupErrorNEIS(Exception):
    pass


def read_api_key():
    try:
        return st.secrets.get("NEIS_API_KEY", "").strip()
    except StreamlitSecretNotFoundError:
        return ""


def last_month():
    first_of_this_month = datetime.now(ZoneInfo("Asia/Seoul")).date().replace(day=1)
    end = first_of_this_month - timedelta(days=1)
    return end.replace(day=1), end


def check_result(result):
    code = result.get("CODE", "")
    if code == "INFO-200":
        return False
    if code != "INFO-000":
        raise LookupErrorNEIS(f"나이스 응답 코드를 확인해 주세요 ({code or '코드 없음'}).")
    return True


@st.cache_data(ttl=600, show_spinner=False)
def fetch_neis(service, filters, api_key):
    page_size = 100 if api_key else 5
    params = {"Type": "json", "pSize": page_size, "pIndex": 1, **filters}
    if api_key:
        params["KEY"] = api_key

    rows = []
    while True:
        try:
            response = requests.get(f"{NEIS_URL}/{service}", params=params, timeout=12)
            response.raise_for_status()
            data = response.json()
        except (requests.RequestException, ValueError):
            # 요청 주소에 KEY가 포함될 수 있어 예외 원문은 화면에 표시하지 않는다.
            raise LookupErrorNEIS("나이스 서버에 연결하거나 응답을 읽는 데 실패했어요.") from None

        try:
            if "RESULT" in data:
                if not check_result(data["RESULT"]) and not rows:
                    return [], 0
                raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            head = data[service][0]["head"]
            for item in head:
                if "RESULT" in item and not check_result(item["RESULT"]):
                    if not rows:
                        return [], 0
                    raise LookupErrorNEIS("요청한 목록을 끝까지 받지 못했어요.")

            total = int(next(item["list_total_count"] for item in head if "list_total_count" in item))
            page = data[service][1]["row"]
            if not isinstance(page, list) or (not page and len(rows) < total):
                raise LookupErrorNEIS("나이스가 보낸 목록이 불완전해요.")
        except (KeyError, IndexError, TypeError, ValueError, StopIteration):
            raise LookupErrorNEIS("나이스 응답에서 목록이나 전체 건수를 읽을 수 없어요.") from None

        rows.extend(page)
        if not api_key or len(rows) >= total:
            return rows, total
        params["pIndex"] += 1


@st.cache_data(ttl=600, show_spinner=False)
def load_teacher_lunch():
    response = requests.get(TEACHER_CSV, timeout=12)
    response.raise_for_status()
    reader = csv.DictReader(StringIO(response.content.decode("utf-8-sig")))
    required = {"MLSV_YMD", "MMEAL_SC_NM", "DDISH_NM"}
    if not required.issubset(reader.fieldnames or []):
        raise ValueError("필수 열 누락")
    return [row for row in reader if row["MMEAL_SC_NM"].strip() == "중식"]


def show_menu(rows, count_label, total):
    by_date = defaultdict(list)
    for row in rows:
        day = datetime.strptime(row["MLSV_YMD"].strip(), "%Y%m%d").date()
        dishes = re.split(r"<br\s*/?\s*>", row["DDISH_NM"], flags=re.IGNORECASE)
        by_date[day].extend(dish.strip() for dish in dishes if dish.strip())

    st.write(f"조회된 날짜 수 {len(by_date)}일 · {count_label} {total}건")
    if not by_date:
        st.info("이 기간에 등록된 중식이 없어요.")
        return

    table = [
        {"날짜": day.isoformat(), "메뉴": " · ".join(by_date[day])}
        for day in sorted(by_date)
    ]
    st.dataframe(table, hide_index=True, use_container_width=True)

def show_ranking(by_date, total, year, month):
    dates_by_dish = defaultdict(set)
    for day, dishes in by_date.items():
        for dish in dishes:
            name = ALLERGY_SUFFIX.sub("", dish).strip()
            if name:
                dates_by_dish[name].add(day)

    if not dates_by_dish:
        st.info("순위를 집계할 메뉴 이름이 없어요.")
        return

    excluded = st.multiselect(
        "반찬 순위에서 뺄 메뉴",
        options=sorted(dates_by_dish),
        help="우리 반에서 빼고 싶은 밥, 국 등의 메뉴를 골라 주세요.",
    )
    included = dates_by_dish.keys() - set(excluded)
    if not included:
        st.info("모든 메뉴를 제외했어요. 제외 목록에서 메뉴를 하나 빼면 순위가 보여요.")
        return

    ranked = sorted(included, key=lambda name: (-len(dates_by_dish[name]), name))
    max_days = len(dates_by_dish[ranked[0]])
    winners = [name for name in ranked if len(dates_by_dish[name]) == max_days]
    first_place = "공동 1위" if len(winners) > 1 else "1위"
    st.write(f"{first_place} · {max_days}일 · {', '.join(winners)}")

    title = (
        "조회된 급식일의 반찬 순위"
        if len(by_date) < total
        else f"{year}년 {month}월 반찬 순위"
    )
    top_ten = ranked[:10]
    chart_rows = [
        {"메뉴": name, "일수": len(dates_by_dish[name]), "표시": f"{len(dates_by_dish[name])}일"}
        for name in top_ten
    ]
    base = alt.Chart(alt.Data(values=chart_rows)).encode(
        y=alt.Y("메뉴:N", sort=top_ten, title=None),
        x=alt.X(
            "일수:Q",
            title="나온 급식일 수",
            axis=alt.Axis(tickMinStep=1, format="d"),
            scale=alt.Scale(domain=[0, max_days + max(1, max_days * 0.2)]),
        ),
        tooltip=["메뉴:N", "일수:Q"],
    )
    bars = base.mark_bar()
    labels = base.mark_text(align="left", baseline="middle", dx=5).encode(text="표시:N")
    chart = (bars + labels).properties(title=title, height=32 * len(top_ten))
    st.altair_chart(chart, use_container_width=True)

    selected = st.selectbox("나온 날짜를 확인할 반찬", options=ranked)
    days = sorted(dates_by_dish[selected])
    st.write(f"{selected} · 총 {len(days)}일")
    st.write(", ".join(day.isoformat() for day in days))

3. 1위를 원자료로 검산하기 · 28~40분

“준비 자료에서 제외 목록을 비우세요. 앱의 날짜 목록을 보기 전에 식단 표에서 배추김치가 나온 날짜를 하나씩 적으세요.” 첫 네 급식일인 10월 1·2·14·15일 가운데 1일과 2일에 나옵니다. 마지막 행까지 확인한 뒤 앱에서 배추김치를 선택합니다. 두 목록의 날짜가 하나씩 대응하는지 확인합니다.

③ 우리 반 기준으로 제외 항목을 바꿔 보세요 · 40~50분

“친환경현미밥과 육개장을 빼면 상위 10개에서 어느 이름이 달라집니까?” 학생용 ‘아롱의 기록’에 학교·기간·제외한 이름과 순위를 적고 남은 반찬의 제공 일수를 확인합니다. 1위가 같으면 그 결과를 기록합니다. 배포 주소와 원래 표에서 확인한 날짜 하나를 함께 남깁니다.

③ 개선 프롬프트
우리 반은 [제외할 메뉴 이름들]을 반찬 순위에서 빼기로 했어. 현재 자료에 있는 이름만 제외 목록에 미리 선택해 줘. 화면에서 선택을 다시 바꿀 수 있게 해 줘. 날짜별 식단과 제공 일수를 세는 방법은 유지하고 고친 main.py 전체를 줘.

예상 질문과 답

“1위는 학생들이 가장 좋아하는 반찬입니까?”

식단에서 확인한 것은 중식을 제공한 날짜와 메뉴입니다. 학생들의 선호도를 알려면 별도의 자료가 필요합니다.

“막대가 열 개면 공동 1위도 열 개까지만 표시합니까?”

막대는 최대 열 개입니다. 공동 1위 명단에는 가장 많이 나온 일수와 같은 이름을 모두 표시합니다.

예상 오류와 대응

학생 화면이나 상황교사의 질문과 조치
비슷한 이름이 따로 나옵니다.원래 표에서 끝의 알레르기 번호만 다른지 확인합니다. ‘찜닭(당면)’의 글자는 남깁니다. ‘간장두부조림’을 두부조림과 합칠지는 반에서 정하고, 합칠 때도 날짜 중복을 확인합니다.
예상한 반찬이 사라졌습니다.제외 목록에서 원래 이름을 찾아 선택을 해제합니다. 모두 제외하면 순위를 그릴 메뉴가 없습니다.
가장 긴 막대가 둘인데 1위가 하나입니다.막대 끝의 최댓값과 같은 일수인 반찬을 모두 읽고 공동 1위 명단과 대조합니다.
키 없이 전체 15건 중 5일만 받았습니다.결론을 조회된 5일로 한정합니다. 월 전체를 조사하려면 인증키로 다시 조회하거나 학교·월이 명시된 준비 자료를 사용합니다.
